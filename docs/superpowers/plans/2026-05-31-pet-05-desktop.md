```typescript
// apps/desktop/src/lib/pet-loader.ts
import type { PetConfig } from "@viben/pet";
import { STANDARD_ANIMATIONS, PET_DEFAULTS } from "@viben/pet";

/**
 * Raw pet.json format from codex-pet-share community
 */
interface RawPetJson {
  id: string;
  displayName: string;
  description: string;
  spritesheetPath: string;
  author?: string;
  tags?: string[];
  source?: string;
  sourceUrl?: string;
}

/**
 * Default accent colors for pets
 */
const DEFAULT_ACCENTS: Record<string, string> = {
  tux: "#f5a623",
  clippit: "#4a90d9",
  dario: "#e74c3c",
  dentist: "#2ecc71",
  "nyako-shigure": "#9b59b6",
  slavik: "#3498db",
  "yelling-dario": "#e74c3c",
  "yorha-sit-2b": "#1a1a2e",
};

/**
 * Default greetings for pets
 */
const DEFAULT_GREETINGS: Record<string, string> = {
  tux: "Hey! Ready to write some code?",
  clippit: "It looks like you're coding. Need help?",
  dario: "Let's build something amazing!",
  dentist: "Time for a code checkup!",
  "nyako-shigure": "Nyaa~ Let me keep you company!",
  slavik: "Privyet! Ready to code?",
  "yelling-dario": "LET'S GOOOOO!!!",
  "yorha-sit-2b": "Glory to mankind.",
};

/**
 * Convert raw pet.json to PetConfig
 */
function rawToPetConfig(raw: RawPetJson, spritesheetUrl: string): PetConfig {
  return {
    id: raw.id,
    name: raw.displayName,
    description: raw.description,
    accent: DEFAULT_ACCENTS[raw.id] ?? "#6366f1",
    greeting: DEFAULT_GREETINGS[raw.id] ?? `Hi! I'm ${raw.displayName}.`,
    spritesheet: spritesheetUrl,
    atlas: {
      cols: 8,
      rows: 9,
      cellWidth: 192,
      cellHeight: 208,
      animations: STANDARD_ANIMATIONS,
    },
    ambient: PET_DEFAULTS.ambient,
    idleTimeoutMs: PET_DEFAULTS.idleTimeoutMs,
  };
}

/**
 * Load pet config from public/pets/ directory
 */
export async function loadPetFromPublic(petId: string): Promise<PetConfig> {
  const configUrl = `/pets/${petId}/pet.json`;

  try {
    const response = await fetch(configUrl);
    if (!response.ok) {
      throw new Error(
        `Failed to load pet "${petId}": ${response.status} ${response.statusText}`
      );
    }

    const raw = (await response.json()) as RawPetJson;
    const spritesheetUrl = `/pets/${petId}/${raw.spritesheetPath}`;

    return rawToPetConfig(raw, spritesheetUrl);
  } catch (err) {
    throw new Error(
      `Error loading pet "${petId}": ${err instanceof Error ? err.message : "Unknown error"}`
    );
  }
}

/**
 * List of available pet IDs
 */
export const AVAILABLE_PETS = [
  "tux",
  "clippit",
  "dario",
  "dentist",
  "nyako-shigure",
  "slavik",
  "yelling-dario",
  "yorha-sit-2b",
] as const;

export type AvailablePetId = (typeof AVAILABLE_PETS)[number];
```

```tsx
// apps/desktop/src/pages/pet-window/index.tsx
import { useEffect, useState } from "react";
import { PetContainer, type PetConfig, type PetPosition } from "@viben/pet";
import { loadPetFromPublic } from "@/lib/pet-loader";

export default function PetWindowPage() {
  const [pet, setPet] = useState<PetConfig | null>(null);
  const [position, setPosition] = useState<PetPosition>({ right: 4, bottom: 4 });
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    // Load default pet (tux)
    loadPetFromPublic("tux")
      .then(setPet)
      .catch((err) => {
        console.error("Failed to load pet:", err);
        setError(err.message);
      })
      .finally(() => setLoading(false));
  }, []);

  if (loading) {
    return (
      <div
        style={{
          width: "100vw",
          height: "100vh",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          background: "transparent",
        }}
      >
        <div style={{ fontSize: "12px", color: "#999" }}>Loading pet...</div>
      </div>
    );
  }

  if (error || !pet) {
    return (
      <div
        style={{
          width: "100vw",
          height: "100vh",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          background: "transparent",
        }}
      >
        <div style={{ fontSize: "12px", color: "#f00" }}>
          Error: {error ?? "Pet not found"}
        </div>
      </div>
    );
  }

  return (
    <div
      style={{
        width: "100vw",
        height: "100vh",
        background: "transparent",
        overflow: "hidden",
      }}
    >
      <PetContainer
        pet={pet}
        position={position}
        onPositionChange={setPosition}
        showBubble={false}
      />
    </div>
  );
}
```

```rust
// apps/desktop/src-tauri/src/commands/pet.rs
use tauri::{AppHandle, Manager};

#[tauri::command]
pub async fn show_pet_window(app: AppHandle) -> Result<(), String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.show().map_err(|e| e.to_string())?;
        window.set_focus().map_err(|e| e.to_string())?;
    }
    Ok(())
}

#[tauri::command]
pub async fn hide_pet_window(app: AppHandle) -> Result<(), String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.hide().map_err(|e| e.to_string())?;
    }
    Ok(())
}

#[tauri::command]
pub async fn toggle_pet_window(app: AppHandle) -> Result<bool, String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        let visible = window.is_visible().map_err(|e| e.to_string())?;
        if visible {
            window.hide().map_err(|e| e.to_string())?;
        } else {
            window.show().map_err(|e| e.to_string())?;
            window.set_focus().map_err(|e| e.to_string())?;
        }
        Ok(!visible)
    } else {
        Ok(false)
    }
}

#[tauri::command]
pub async fn is_pet_window_visible(app: AppHandle) -> Result<bool, String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.is_visible().map_err(|e| e.to_string())
    } else {
        Ok(false)
    }
}
```

## Summary

This plan integrates the pet system into the desktop app with a standalone entry point:

| Component | Purpose |
|-----------|---------|
| `pet-window.html` | HTML entry point (like screenshot-overlay.html) |
| `pet-window-main.tsx` | React entry point (like screenshot-overlay-main.tsx) |
| `pages/pet-window/index.tsx` | Pet window page component |
| `lib/pet-loader.ts` | Loader utility for public/pets/ |
| `commands/pet.rs` | Window management commands |
| Tray menu | Show/hide control |
| `public/pets/` | Pet presets (8 presets) |

**Architecture**:
- Pet window is completely independent (separate HTML/TSX entry)
- Loads pets from `apps/desktop/public/pets/`
- No route in main App.tsx - standalone window
- Similar pattern to screenshot-overlay
