```tsx
// apps/desktop/src/pages/pet/index.tsx
import { useEffect, useState } from 'react';
import { 
  PetContainer,
  type PetConfig,
  type PetPosition,
  PET_DEFAULTS,
} from '@viben/pet';
import '@viben/pet/styles/pet.css';

// For now, use a bundled demo pet config
// Later this will load from presets via the store
const DEMO_PET: PetConfig = {
  id: 'viben-mascot',
  name: 'Viben',
  description: 'Viben 官方吉祥物',
  accent: '#6366f1',
  greeting: '嗨！我是 Viben，随时准备陪你一起创造！',
  spritesheet: '/pets/viben-mascot/spritesheet.webp',
  atlas: {
    cols: 8,
    rows: 9,
    cellWidth: 192,
    cellHeight: 208,
    animations: [
      { id: 'idle', row: 0, frames: 6, fps: 6 },
      { id: 'running-right', row: 1, frames: 8, fps: 8 },
      { id: 'running-left', row: 2, frames: 8, fps: 8 },
      { id: 'waving', row: 3, frames: 4, fps: 6 },
      { id: 'jumping', row: 4, frames: 5, fps: 7 },
      { id: 'failed', row: 5, frames: 8, fps: 7 },
      { id: 'waiting', row: 6, frames: 6, fps: 6 },
      { id: 'running', row: 7, frames: 6, fps: 8 },
      { id: 'review', row: 8, frames: 6, fps: 6 },
    ],
  },
  ambient: PET_DEFAULTS.ambient,
  idleTimeoutMs: PET_DEFAULTS.idleTimeoutMs,
};

export default function PetPage() {
  const [pet, setPet] = useState<PetConfig | null>(DEMO_PET);
  const [position, setPosition] = useState<PetPosition>({ right: 0, bottom: 0 });

  // In transparent window mode, position is relative to window, not viewport
  // The window itself is positioned by Tauri
  useEffect(() => {
    // Center the pet in the window
    setPosition({ right: 4, bottom: 4 });
  }, []);

  if (!pet) {
    return null;
  }

  return (
    <div 
      style={{ 
        width: '100vw', 
        height: '100vh',
        background: 'transparent',
        overflow: 'hidden',
      }}
    >
      <PetContainer
        pet={pet}
        position={position}
        onPositionChange={setPosition}
        showBubble={false}
      />
    </div>
  );
}
```

```typescript
// apps/desktop/src/stores/pet-store.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import type { PetConfig, PetPosition } from '@viben/pet';
import { PET_DEFAULTS } from '@viben/pet';

interface DesktopPetState {
  // Pet state
  enabled: boolean;
  pet: PetConfig | null;
  position: PetPosition;
  
  // Actions
  setEnabled: (enabled: boolean) => void;
  setPet: (pet: PetConfig | null) => void;
  setPosition: (position: PetPosition) => void;
  reset: () => void;
}

const initialState = {
  enabled: true,
  pet: null,
  position: PET_DEFAULTS.position,
};

export const useDesktopPetStore = create<DesktopPetState>()(
  persist(
    (set) => ({
      ...initialState,
      
      setEnabled: (enabled) => set({ enabled }),
      setPet: (pet) => set({ pet }),
      setPosition: (position) => set({ position }),
      reset: () => set(initialState),
    }),
    {
      name: 'viben:desktop-pet',
      storage: createJSONStorage(() => localStorage),
    }
  )
);
```

If the app doesn't have generic window show/hide commands, create them:

```rust
// apps/desktop/src-tauri/src/commands/pet.rs
use tauri::{AppHandle, Manager};

#[tauri::command]
pub async fn show_pet_window(app: AppHandle) -> Result<(), String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.show().map_err(|e| e.to_string())?;
        window.set_focus().map_err(|e| e.to_string())?;
    }
    Ok(())
}

#[tauri::command]
pub async fn hide_pet_window(app: AppHandle) -> Result<(), String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.hide().map_err(|e| e.to_string())?;
    }
    Ok(())
}

#[tauri::command]
pub async fn toggle_pet_window(app: AppHandle) -> Result<bool, String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        let visible = window.is_visible().map_err(|e| e.to_string())?;
        if visible {
            window.hide().map_err(|e| e.to_string())?;
        } else {
            window.show().map_err(|e| e.to_string())?;
            window.set_focus().map_err(|e| e.to_string())?;
        }
        Ok(!visible)
    } else {
        Ok(false)
    }
}

#[tauri::command]
pub async fn move_pet_window(app: AppHandle, x: i32, y: i32) -> Result<(), String> {
    if let Some(window) = app.get_webview_window("pet-window") {
        window.set_position(tauri::Position::Physical(
            tauri::PhysicalPosition::new(x, y)
        )).map_err(|e| e.to_string())?;
    }
    Ok(())
}
```
