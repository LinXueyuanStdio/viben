```typescript
// packages/pet/src/atlas.ts
import {
  CODEX_ATLAS,
  STANDARD_ANIMATIONS,
  type PetAnimationDef,
  type PetAtlasLayout,
} from './types';

/** Aspect ratio tolerance for atlas detection (6%) */
const ASPECT_TOLERANCE = 0.06;

/** Fallback order when requested animation is missing */
const ROW_FALLBACK_ORDER = ['idle', 'waiting', 'waving', 'running', 'running-right'];

/**
 * Check if dimensions match Codex atlas aspect ratio.
 * Accepts proportionally scaled atlases.
 */
export function looksLikeCodexAtlas(width: number, height: number): boolean {
  if (!Number.isFinite(width) || !Number.isFinite(height)) return false;
  if (width <= 0 || height <= 0) return false;
  const aspect = width / height;
  return Math.abs(aspect - CODEX_ATLAS.aspect) < ASPECT_TOLERANCE;
}

/**
 * Get animation definition by ID.
 */
export function getAnimationDef(
  animations: PetAnimationDef[],
  id: string
): PetAnimationDef | undefined {
  return animations.find((a) => a.id === id);
}

/**
 * Pick the best animation row, with fallback chain.
 * Returns undefined only if animations array is empty.
 */
export function pickAnimationRow(
  animations: PetAnimationDef[],
  preferred: string
): PetAnimationDef | undefined {
  if (animations.length === 0) return undefined;

  // Try exact match first
  const direct = getAnimationDef(animations, preferred);
  if (direct) return direct;

  // Walk fallback chain
  for (const id of ROW_FALLBACK_ORDER) {
    const fallback = getAnimationDef(animations, id);
    if (fallback) return fallback;
  }

  // Last resort: first available
  return animations[0];
}

/**
 * Create standard 8x9 Codex atlas layout.
 */
export function createStandardAtlasLayout(
  animations: PetAnimationDef[] = STANDARD_ANIMATIONS
): PetAtlasLayout {
  return {
    cols: CODEX_ATLAS.cols,
    rows: CODEX_ATLAS.rows,
    cellWidth: CODEX_ATLAS.cellWidth,
    cellHeight: CODEX_ATLAS.cellHeight,
    animations,
  };
}

/**
 * Calculate CSS background-position for a specific frame.
 */
export function getFramePosition(
  layout: PetAtlasLayout,
  row: number,
  frame: number
): { x: number; y: number } {
  return {
    x: -(frame * layout.cellWidth),
    y: -(row * layout.cellHeight),
  };
}

/**
 * Validate atlas layout has required properties.
 */
export function validateAtlasLayout(layout: unknown): layout is PetAtlasLayout {
  if (!layout || typeof layout !== 'object') return false;
  const l = layout as Record<string, unknown>;
  return (
    typeof l.cols === 'number' &&
    typeof l.rows === 'number' &&
    typeof l.cellWidth === 'number' &&
    typeof l.cellHeight === 'number' &&
    Array.isArray(l.animations) &&
    l.cols > 0 &&
    l.rows > 0 &&
    l.cellWidth > 0 &&
    l.cellHeight > 0
  );
}
```

```typescript
// packages/pet/src/interaction.ts
import type { PetInteraction } from './types';

/** Minimum drag distance to register a gesture (px) */
export const DRAG_GESTURE_MIN_PX = 14;

/** Axis dominance ratio to classify direction */
export const DRAG_AXIS_BIAS = 1.18;

/** Map interaction state to animation ID */
export const INTERACTION_TO_ANIMATION: Record<PetInteraction, string> = {
  idle: 'idle',
  hover: 'waving',
  'drag-right': 'running-right',
  'drag-left': 'running-left',
  'drag-up': 'jumping',
  'drag-down': 'waving',
  waiting: 'waiting',
};

/**
 * Get the animation ID for an interaction state.
 */
export function getAnimationIdForInteraction(interaction: PetInteraction): string {
  return INTERACTION_TO_ANIMATION[interaction];
}

/**
 * Classify drag direction from delta movement.
 * Returns null if movement is below threshold or diagonal.
 */
export function classifyDragDirection(
  dx: number,
  dy: number,
  minPx: number = DRAG_GESTURE_MIN_PX,
  axisBias: number = DRAG_AXIS_BIAS
): PetInteraction | null {
  const absX = Math.abs(dx);
  const absY = Math.abs(dy);

  // Below threshold
  if (absX < minPx && absY < minPx) return null;

  // Check axis dominance
  if (absX >= absY * axisBias) {
    return dx > 0 ? 'drag-right' : 'drag-left';
  }
  if (absY >= absX * axisBias) {
    return dy < 0 ? 'drag-up' : 'drag-down';
  }

  // Diagonal, no clear direction
  return null;
}

/**
 * Determine next interaction state based on current state and event.
 */
export function resolveInteraction(
  current: PetInteraction,
  event: 'pointerenter' | 'pointerleave' | 'pointerup' | 'idle-timeout',
  isDragging: boolean
): PetInteraction {
  // During drag, interaction is controlled by drag direction
  if (isDragging) return current;

  switch (event) {
    case 'pointerenter':
      return 'hover';
    case 'pointerleave':
      return 'idle';
    case 'pointerup':
      return 'idle';
    case 'idle-timeout':
      return current === 'idle' ? 'waiting' : current;
    default:
      return current;
  }
}
```

```typescript
import type { PetAnimationDef, PetAtlasLayout } from './types';
import { getAnimationDef } from './atlas';

/**
 * Generate random value in range [min, min + variance].
 */
export function randomInRange(config: { min: number; variance: number }): number {
  return config.min + Math.floor(Math.random() * (config.variance + 1));
}

/**
 * Pick a random ambient animation from the pool.
 * Avoids repeating the last played animation when possible.
 */
export function pickAmbientAnimation(
  animations: PetAnimationDef[],
  pool: string[],
  avoidId?: string
): PetAnimationDef | null {
  // Filter to valid animations in pool
  const valid = animations.filter((a) => pool.includes(a.id));
  if (valid.length === 0) return null;

  // Try to avoid last played
  const candidates =
    valid.length > 1 && avoidId
      ? valid.filter((a) => a.id !== avoidId)
      : valid;

  const choices = candidates.length > 0 ? candidates : valid;
  return choices[Math.floor(Math.random() * choices.length)] ?? null;
}

/**
 * Animation style properties for CSS.
 */
export interface AnimationStyle {
  width: number;
  height: number;
  backgroundSize: string;
  backgroundPositionY: number;
  animationDuration: string;
  animationTimingFunction: string;
}

/**
 * Create CSS style object for sprite animation.
 */
export function createAnimationStyle(
  layout: PetAtlasLayout,
  animation: PetAnimationDef
): AnimationStyle {
  const totalWidth = layout.cols * layout.cellWidth;
  const totalHeight = layout.rows * layout.cellHeight;
  const duration = animation.frames / animation.fps;

  return {
    width: layout.cellWidth,
    height: layout.cellHeight,
    backgroundSize: `${totalWidth}px ${totalHeight}px`,
    backgroundPositionY: -(animation.row * layout.cellHeight),
    animationDuration: `${duration.toFixed(3).replace(/\.?0+$/, '')}s`,
    animationTimingFunction: `steps(${animation.frames})`,
  };
}

/**
 * Generate CSS keyframes string for horizontal sprite animation.
 * The animation moves from frame 0 to the last frame.
 */
export function generateSpriteKeyframes(
  layout: PetAtlasLayout,
  animation: PetAnimationDef
): string {
  const endX = -(animation.frames * layout.cellWidth);
  return `
    from { background-position-x: 0; }
    to { background-position-x: ${endX}px; }
  `;
}

/**
 * Calculate the frame index at a given time.
 */
export function getFrameAtTime(
  animation: PetAnimationDef,
  timeMs: number
): number {
  const frameDuration = 1000 / animation.fps;
  const frame = Math.floor(timeMs / frameDuration) % animation.frames;
  return frame;
}
```

```typescript
// packages/pet/src/store.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import type { PetConfig, PetInteraction, PetPosition } from './types';
import { PET_DEFAULTS } from './types';

export interface PetState {
  /** Current active pet config */
  pet: PetConfig | null;
  /** Position relative to viewport bottom-right */
  position: PetPosition;
  /** Current interaction state */
  interaction: PetInteraction;
  /** Currently playing ambient animation ID */
  ambientRowId: string | null;
  /** Whether speech bubble is open */
  bubbleOpen: boolean;

  // Actions
  setPet: (pet: PetConfig | null) => void;
  setPosition: (position: PetPosition) => void;
  setInteraction: (interaction: PetInteraction) => void;
  setAmbientRowId: (id: string | null) => void;
  setBubbleOpen: (open: boolean) => void;
  reset: () => void;
}

const initialState = {
  pet: null,
  position: PET_DEFAULTS.position,
  interaction: 'idle' as PetInteraction,
  ambientRowId: null,
  bubbleOpen: false,
};

/**
 * Create a pet store instance with optional persistence.
 * Use unique storageKey for testing or multiple instances.
 */
export function createPetStore(storageKey = 'viben:pet') {
  return create<PetState>()(
    persist(
      (set) => ({
        ...initialState,

        setPet: (pet) => set({ pet }),
        setPosition: (position) => set({ position }),
        setInteraction: (interaction) => set({ interaction }),
        setAmbientRowId: (ambientRowId) => set({ ambientRowId }),
        setBubbleOpen: (bubbleOpen) => set({ bubbleOpen }),
        reset: () => set(initialState),
      }),
      {
        name: storageKey,
        storage: createJSONStorage(() => {
          // Use in-memory storage for SSR/testing
          if (typeof window === 'undefined') {
            const store = new Map<string, string>();
            return {
              getItem: (key) => store.get(key) ?? null,
              setItem: (key, value) => store.set(key, value),
              removeItem: (key) => store.delete(key),
            };
          }
          return localStorage;
        }),
        // Only persist pet and position, not transient state
        partialize: (state) => ({
          pet: state.pet,
          position: state.position,
        }),
      }
    )
  );
}

/** Default store instance for application use */
export const usePetStore = createPetStore();
```
