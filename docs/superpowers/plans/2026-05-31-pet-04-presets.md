// packages/core/templates/viben/pets/registry.ts
import type { PetConfig, PetPreset, PetAtlasLayout } from '@viben/pet';
import { STANDARD_ANIMATIONS, PET_DEFAULTS } from '@viben/pet';

/**
 * Raw pet.json format from codex-pet-share
 */
interface RawPetJson {
  id: string;
  displayName: string;
  description: string;
  spritesheetPath: string;
  author?: string;
  tags?: string[];
  source?: string;
  sourceUrl?: string;
}

/**
 * Get preset by ID.
 */
export function getPresetById(id: string): PetPreset | undefined {
  return PET_PRESETS.find((p) => p.id === id);
}

/**
 * Convert raw pet.json to PetConfig.
 */
function rawToPetConfig(raw: RawPetJson, spritesheetUrl: string): PetConfig {
  return {
    id: raw.id,
    name: raw.displayName,
    description: raw.description,
    accent: DEFAULT_ACCENTS[raw.id] ?? '#6366f1',
    greeting: DEFAULT_GREETINGS[raw.id] ?? `Hi! I'm ${raw.displayName}.`,
    spritesheet: spritesheetUrl,
    atlas: CODEX_ATLAS_LAYOUT,
    ambient: PET_DEFAULTS.ambient,
    idleTimeoutMs: PET_DEFAULTS.idleTimeoutMs,
  };
}

/**
 * Load full pet config from a preset.
 * @param preset - The preset to load
 * @param baseUrl - Base URL for resolving paths (e.g., '/pets/' or 'http://...')
 */
export async function loadPresetConfig(
  preset: PetPreset,
  baseUrl: string = ''
): Promise<PetConfig> {
  const configUrl = `${baseUrl}${preset.path}/pet.json`;

  const response = await fetch(configUrl);
  if (!response.ok) {
    throw new Error(`Failed to load pet config: ${response.status}`);
  }

  const raw = (await response.json()) as RawPetJson;
  const spritesheetUrl = `${baseUrl}${preset.path}/${raw.spritesheetPath}`;

  return rawToPetConfig(raw, spritesheetUrl);
}

/**
 * Load preset by ID.
 */
export async function loadPresetById(
  id: string,
  baseUrl: string = ''
): Promise<PetConfig> {
  const preset = getPresetById(id);
  if (!preset) {
    throw new Error(`Unknown preset: ${id}`);
  }
  return loadPresetConfig(preset, baseUrl);
}

/**
 * Get all preset IDs.
 */
export function getPresetIds(): string[] {
  return PET_PRESETS.map((p) => p.id);
}
