- [ ] **Step 3: Create App.tsx**

```tsx
import { useState, useCallback } from 'react';
import {
  PetContainer,
  usePetStore,
  type PetConfig,
  type PetPosition,
  PET_DEFAULTS,
  STANDARD_ANIMATIONS,
} from '@viben/pet';

// Demo pet config - will be replaced with real preset
const DEMO_PET: PetConfig = {
  id: 'demo-pet',
  name: 'Demo',
  description: 'A demo pet for testing',
  accent: '#6366f1',
  greeting: 'Hi! I am a demo pet. Try dragging me around!',
  spritesheet: '/assets/demo-pet/spritesheet.png',
  atlas: {
    cols: 8,
    rows: 9,
    cellWidth: 192,
    cellHeight: 208,
    animations: STANDARD_ANIMATIONS,
  },
  ambient: PET_DEFAULTS.ambient,
  idleTimeoutMs: PET_DEFAULTS.idleTimeoutMs,
};

export function App() {
  const [pet, setPet] = useState<PetConfig | null>(null);
  const [position, setPosition] = useState<PetPosition>(PET_DEFAULTS.position);
  const [showBubble, setShowBubble] = useState(true);
  const [loading, setLoading] = useState(true);

  // Load demo pet on mount
  useEffect(() => {
    loadDemoPet()
      .then(setPet)
      .catch(console.error)
      .finally(() => setLoading(false));
  }, []);

  const handleTap = useCallback(() => {
    setShowBubble((prev) => !prev);
  }, []);

  const handleReset = useCallback(() => {
    setPosition(PET_DEFAULTS.position);
  }, []);

  const handleTogglePet = useCallback(() => {
    if (pet) {
      setPet(null);
    } else {
      loadDemoPet().then(setPet);
    }
  }, [pet]);

  if (loading) {
    return (
      <div className="example-app">
        <div style={{ textAlign: 'center', padding: '100px', color: 'white' }}>
          Loading demo pet...
        </div>
      </div>
    );
  }

  return (
    <div className="example-app">
      <header className="example-header">
        <h1>Viben Pet Development</h1>
        <p>Interactive testing environment for @viben/pet components</p>
      </header>

      {/* Debug Panel */}
      <div className="debug-panel">
        <h3>Debug Info</h3>
        <pre>
          {JSON.stringify(
            {
              pet: pet?.id ?? null,
              position,
              showBubble,
              loading,
            },
            null,
            2
          )}
        </pre>
        <div className="debug-controls">
          <button onClick={handleReset}>Reset Position</button>
          <button onClick={handleTogglePet}>
            {pet ? 'Hide Pet' : 'Show Pet'}
          </button>
          <button onClick={() => setShowBubble((p) => !p)}>
            Toggle Bubble
          </button>
        </div>
      </div>

      {/* Instructions */}
      <div className="instructions">
        <strong>Interactions:</strong>
        Click to toggle bubble
        <br />
        Drag to move
        <br />
        Hover to wave
        <br />
        Wait 45s for idle animation
      </div>

      {/* Pet Container */}
      <PetContainer
        pet={pet}
        position={position}
        onPositionChange={setPosition}
        onTap={handleTap}
        showBubble={showBubble}
      />
    </div>
  );
}
```
